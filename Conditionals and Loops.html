    • It returns something positive if a comes before b
    • It returns something negative if b comes before a
  • Java also has a boolean variable type and keywords true and false
  • You can create more complex statements use logical operators: “and” and “or”
    • The and operators is &&
    • The or operator is ||
    • The negation operator flips the truth value of an expression or boolean variable: !

Loops

  • Java has for loops and while loops
package unl.soc;

public class Demo {
    public static void main(String args[]) {
    
        int n = 10;
        
        for (int i=0; i<n; i++) {
            System.out.println(i);
        }
        System.out.println("Done");
        
        int i = 0;
        while (i<n) {
            System.out.println(i);
            i++;
        }
        
        int x = 1234;
        int numDigits = 0;
        // determine how many digits are in x...
        if (x == 0){
            numDigits = 0;
        } else {
            while (x > 0) {
                numDigits++;
                x /= 10;
            }
        }
        System.out.println("There are " + numDigits + " digits");
        
        int primes[] = [2, 3, 5, 7, 11, 13, 17];
        for (int j = 0; j <primes.length; j++) {
            System.out.println(primes[j]);
        }
        
        
        // for each
        for (int z: primes) {
            System.out.println(z);
        }
    }
}
