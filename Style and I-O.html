Style Notes

  • ALL variable names should be lowerCamelCase and nouns (pluralize collections only)
  • All methods names should be lowerCamelCase and should be verbs
  • ALL class names should be UpperCamelCase, never pluralized, and nouns
  • Use proper Java-style whitespace

File I/O

File Output

  • Open the file, write to the file, close the file
    • failure to close the file may lead to data corruption
package unl.soc;

import.io.File;

public class Demo {
    public static void main(String args[]) {
    
        int a = 10;
        double b = 3.5;
        
        //file output:
        File f = new File("data/myOutput.txt");
        PrintWriter pw = new PrintWriter(f);
        try {
            pw.println("Hello, this is one line");
            pw.print("This is aloien without an endline");
            pw.print("now I'll put the endline in: \n");
            pw.printf("Variables: %d, %f\n", a, b);
            pw.close();
        } catch (FileNotFoundException e) {
            e.printStackTrace();
        }
    }
}
